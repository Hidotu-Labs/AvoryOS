cmake_minimum_required(VERSION 3.16)
project(NetworkIconProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml)
find_package(KF6NetworkManagerQt REQUIRED)
find_package(KF6ModemManagerQt REQUIRED)
find_package(KF6Config REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MM REQUIRED IMPORTED_TARGET mm-glib)
set(SRC "${PLASMA_NM_SOURCE}")
add_executable(icon-probe main.cpp ${SRC}/libs/connectionicon.cpp ${SRC}/libs/connectionicon.h)
target_include_directories(icon-probe PRIVATE ${SRC}/libs ${SRC}/libs/editor ${PLASMA_NM_BUILD}/libs/editor)
target_link_libraries(icon-probe PRIVATE Qt6::Core Qt6::Qml KF6::NetworkManagerQt KF6::ModemManagerQt KF6::ConfigCore PkgConfig::MM /usr/lib/libplasmanm_editor.so)
add_executable(icon-module-probe module.cpp)
target_link_libraries(icon-module-probe PRIVATE Qt6::Core Qt6::Qml)
