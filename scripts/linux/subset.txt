# Paths copied verbatim from the pinned Linux tree into kernel/linux/.
#
# One path per line; directories are copied recursively.  Keep this list as
# small as possible and grow it when a phase actually needs more source.
# Entries are relative to the Linux tree root.

# Phase 0: enough headers to compile standalone library files and the first
# LinuxKPI glue.  Phase 1+ adds lib/, drivers/gpu/drm/, drivers/dma-buf/, ...
include
arch/x86/include
lib/math/div64.c

# Phase 1: pure algorithm libraries with no kernel-integration dependencies.
lib/find_bit.c
lib/list_sort.c
lib/math/gcd.c
lib/math/int_sqrt.c

# Phase 1 containers and primitives on top of the LinuxKPI slab/lock/RCU
# cores.  lib/radix-tree.h is the private header xarray.c includes by name.
lib/xarray.c
lib/refcount.c
lib/siphash.c
lib/radix-tree.h
lib/radix-tree.c
lib/idr.c

# Phase 1 remaining library imports.  lib/crc32.c needs its private
# crc32defs.h; linux-import.sh generates the slicing-by-8 crc32table.h.
lib/math/rational.c
lib/math/reciprocal_div.c
lib/crc32.c
lib/crc32defs.h
lib/xxhash.c
lib/kfifo.c

# Full bitmap import (replaces the temporary bitmap_compat.c shim), its
# _parse_integer dependency from kstrtox.c, and string_helpers.c (which needs
# the shared hex_asc/_ctype tables).
lib/bitmap.c
lib/kstrtox.c
lib/kstrtox.h
lib/string_helpers.c
lib/hexdump.c
lib/ctype.c

# Kbuild generates <generated/timeconst.h> from HZ; the import script derives
# it from this script instead of hand-writing constants.
kernel/time/timeconst.bc

# Phase 2: DMA-buf/fence/resv/sync_file stack.  The whole directory is copied
# so private headers (dma-buf-sysfs-stats.h, sync_debug.h, ...) are present;
# scripts/linux/files.txt decides which .c files actually build.
drivers/dma-buf

# dma-fence-unwrap.c calls sort(); import the self-contained implementation.
lib/sort.c

# Phase 3: DRM core headers and sources.  The root glob copies drm.ko's and
# drm_kms_helper.ko's files plus their private headers without dragging in
# every vendor subdirectory; the canary drivers are listed explicitly.  amdgpu,
# the display helpers and the scheduler arrive with the phases that need them.
drivers/gpu/drm/*.h
drivers/gpu/drm/*.c
drivers/gpu/drm/vgem
drivers/gpu/drm/vkms
drivers/gpu/drm/tiny

# Phase 4: TTM and the GPU scheduler.  Public headers already come with the
# include/ copy; the private ttm_module.h and scheduler/gpu_scheduler_trace.h
# live in the source directories, so the whole directories are copied.
drivers/gpu/drm/ttm
drivers/gpu/drm/scheduler

# Phase 6: the amdgpu driver tree (amdgpu/display/pm/include, including the
# asic_reg register headers, ~400 MB in the gitignored import) and the
# drm_display_helper sources amdgpu selects.  The amdgpu object list itself is
# generated from the Kbuild fragments by scripts/linux/kbuild-subset.py, not
# hand-maintained in files.txt.
drivers/gpu/drm/amd
drivers/gpu/drm/display

# Phase 6: i2c-algo-bit, used by amdgpu_i2c.c for DDC/EDID bit-banging.
drivers/i2c/algos/i2c-algo-bit.c

# Phase 6 C2: HDMI infoframe packing (drivers/video/hdmi.c) — the DRM HDMI
# helper and amdgpu's DM link code link against hdmi_*_infoframe_*().
drivers/video/hdmi.c
